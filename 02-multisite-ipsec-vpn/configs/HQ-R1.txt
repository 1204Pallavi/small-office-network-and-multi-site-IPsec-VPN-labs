! HQ-R1 - headquarters edge router and VPN hub. Cisco 2911 (securityk9 licence required).
! Roles: GRE-over-IPsec to BR1 and BR2, OSPF over tunnels, DHCP server for HQ VLANs.
!
enable
configure terminal
hostname HQ-R1
no ip domain-lookup
ip domain-name hq.lab
enable secret CHANGE_ME
username admin secret CHANGE_ME
service password-encryption
!
interface GigabitEthernet0/0
 description WAN to ISP
 ip address 203.0.113.2 255.255.255.252
 no shutdown
interface GigabitEthernet0/1
 description Routed link to DS1
 ip address 10.1.0.1 255.255.255.252
 no shutdown
interface GigabitEthernet0/2
 description Routed link to DS2
 ip address 10.1.0.5 255.255.255.252
 no shutdown
!
! --- IKE phase 1 ---
crypto isakmp policy 10
 encryption aes 256
 hash sha
 authentication pre-share
 group 2
 lifetime 86400
crypto isakmp key CHANGE_ME address 198.51.100.2
crypto isakmp key CHANGE_ME address 192.0.2.2
!
! --- IPsec phase 2 ---
crypto ipsec transform-set VPN-TS esp-aes 256 esp-sha-hmac
!
! Crypto ACLs: protect the GRE traffic between tunnel endpoints
access-list 101 permit gre host 203.0.113.2 host 198.51.100.2
access-list 102 permit gre host 203.0.113.2 host 192.0.2.2
!
crypto map VPN-MAP 10 ipsec-isakmp
 set peer 198.51.100.2
 set transform-set VPN-TS
 match address 101
crypto map VPN-MAP 20 ipsec-isakmp
 set peer 192.0.2.2
 set transform-set VPN-TS
 match address 102
!
interface GigabitEthernet0/0
 crypto map VPN-MAP
!
! --- GRE tunnels ---
interface Tunnel0
 description GRE/IPsec to BR1-R1
 ip address 172.16.1.1 255.255.255.252
 tunnel source GigabitEthernet0/0
 tunnel destination 198.51.100.2
interface Tunnel1
 description GRE/IPsec to BR2-R1
 ip address 172.16.2.1 255.255.255.252
 tunnel source GigabitEthernet0/0
 tunnel destination 192.0.2.2
!
! --- Routing ---
router ospf 1
 router-id 1.1.1.1
 network 10.1.0.0 0.0.0.7 area 0
 network 172.16.1.0 0.0.0.3 area 0
 network 172.16.2.0 0.0.0.3 area 0
 passive-interface GigabitEthernet0/0
!
ip route 0.0.0.0 0.0.0.0 203.0.113.1
!
! --- DHCP for HQ VLANs (relayed by DS1/DS2 via ip helper-address) ---
ip dhcp excluded-address 10.1.10.1 10.1.10.10
ip dhcp excluded-address 10.1.20.1 10.1.20.10
ip dhcp pool ENG
 network 10.1.10.0 255.255.255.0
 default-router 10.1.10.1
ip dhcp pool OPS
 network 10.1.20.0 255.255.255.0
 default-router 10.1.20.1
!
! --- SSH ---
crypto key generate rsa modulus 2048
ip ssh version 2
line vty 0 4
 transport input ssh
 login local
line console 0
 logging synchronous
end
write memory
