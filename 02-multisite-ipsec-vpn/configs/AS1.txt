! AS1 - HQ access switch. Cisco 2960-24TT.
! Roles: RSTP, dual uplinks, DHCP snooping, sticky port security, PortFast + BPDU Guard.
!
enable
configure terminal
hostname AS1
no ip domain-lookup
enable secret CHANGE_ME
service password-encryption
!
spanning-tree mode rapid-pvst
!
vlan 10
 name ENG
vlan 20
 name OPS
vlan 999
 name BLACKHOLE
!
! --- DHCP snooping (the DHCP server is reached through the trusted uplinks) ---
ip dhcp snooping
ip dhcp snooping vlan 10,20
no ip dhcp snooping information option
!
! --- Uplinks ---
interface FastEthernet0/1
 description Trunk to DS1 Fa0/1
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20
 ip dhcp snooping trust
 no shutdown
interface FastEthernet0/2
 description Trunk to DS2 Fa0/1
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20
 ip dhcp snooping trust
 no shutdown
!
! --- ENG user ports ---
interface range FastEthernet0/3 - 12
 description ENG users
 switchport mode access
 switchport access vlan 10
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation shutdown
 spanning-tree portfast
 spanning-tree bpduguard enable
 ip dhcp snooping limit rate 15
 no shutdown
!
! --- OPS user ports ---
interface range FastEthernet0/13 - 22
 description OPS users
 switchport mode access
 switchport access vlan 20
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation shutdown
 spanning-tree portfast
 spanning-tree bpduguard enable
 ip dhcp snooping limit rate 15
 no shutdown
!
! --- Unused ports ---
interface range FastEthernet0/23 - 24
 description UNUSED
 switchport mode access
 switchport access vlan 999
 shutdown
!
line console 0
 logging synchronous
end
write memory
