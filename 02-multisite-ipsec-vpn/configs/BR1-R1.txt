! BR1-R1 - Branch 1 router. Cisco 2911 (securityk9 licence required).
! Roles: GRE-over-IPsec spoke to HQ, OSPF, DHCP server for the branch LAN.
!
enable
configure terminal
hostname BR1-R1
no ip domain-lookup
ip domain-name br1.lab
enable secret CHANGE_ME
username admin secret CHANGE_ME
service password-encryption
!
interface GigabitEthernet0/0
 description WAN to ISP
 ip address 198.51.100.2 255.255.255.252
 no shutdown
interface GigabitEthernet0/1
 description Branch 1 LAN
 ip address 10.2.10.1 255.255.255.0
 no shutdown
!
! --- IKE phase 1 ---
crypto isakmp policy 10
 encryption aes 256
 hash sha
 authentication pre-share
 group 2
 lifetime 86400
crypto isakmp key CHANGE_ME address 203.0.113.2
!
! --- IPsec phase 2 ---
crypto ipsec transform-set VPN-TS esp-aes 256 esp-sha-hmac
access-list 101 permit gre host 198.51.100.2 host 203.0.113.2
crypto map VPN-MAP 10 ipsec-isakmp
 set peer 203.0.113.2
 set transform-set VPN-TS
 match address 101
!
interface GigabitEthernet0/0
 crypto map VPN-MAP
!
! --- GRE tunnel ---
interface Tunnel0
 description GRE/IPsec to HQ-R1
 ip address 172.16.1.2 255.255.255.252
 tunnel source GigabitEthernet0/0
 tunnel destination 203.0.113.2
!
! --- Routing ---
router ospf 1
 router-id 3.3.3.3
 network 172.16.1.0 0.0.0.3 area 0
 network 10.2.10.0 0.0.0.255 area 0
 passive-interface GigabitEthernet0/0
 passive-interface GigabitEthernet0/1
!
ip route 0.0.0.0 0.0.0.0 198.51.100.1
!
! --- DHCP for the branch LAN ---
ip dhcp excluded-address 10.2.10.1 10.2.10.10
ip dhcp pool BR1-LAN
 network 10.2.10.0 255.255.255.0
 default-router 10.2.10.1
!
! --- SSH ---
crypto key generate rsa modulus 2048
ip ssh version 2
line vty 0 4
 transport input ssh
 login local
line console 0
 logging synchronous
end
write memory
