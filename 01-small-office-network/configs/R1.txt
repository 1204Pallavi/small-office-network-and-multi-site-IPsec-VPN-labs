! R1 - edge router. Cisco 2911.
! Roles: NAT/PAT, static NAT for the web server, OSPF default originate, inbound ACL on WAN, SSH.
!
enable
configure terminal
hostname R1
no ip domain-lookup
ip domain-name office.lab
enable secret CHANGE_ME
username admin secret CHANGE_ME
service password-encryption
banner motd # Authorised access only #
!
interface GigabitEthernet0/0
 description WAN to ISP
 ip address 203.0.113.2 255.255.255.248
 ip nat outside
 ip access-group OUTSIDE-IN in
 no shutdown
!
interface GigabitEthernet0/1
 description LAN to CORE-SW
 ip address 10.10.0.1 255.255.255.252
 ip nat inside
 no shutdown
!
! --- Routing ---
router ospf 1
 router-id 1.1.1.1
 network 10.10.0.0 0.0.0.3 area 0
 passive-interface GigabitEthernet0/0
 default-information originate
!
ip route 0.0.0.0 0.0.0.0 203.0.113.1
!
! --- NAT ---
access-list 1 permit 10.10.0.0 0.0.255.255
ip nat inside source list 1 interface GigabitEthernet0/0 overload
ip nat inside source static 10.10.30.10 203.0.113.3
!
! --- Inbound filtering on the WAN interface ---
! (evaluated before NAT for inbound packets, so the web server is matched by its public address)
ip access-list extended OUTSIDE-IN
 permit tcp any host 203.0.113.3 eq www
 permit icmp any any echo-reply
 permit icmp any any unreachable
 permit icmp any any time-exceeded
 permit tcp any any established
 permit udp any eq domain any
 deny ip any any log
!
! --- Management: SSH only, from the management VLAN ---
crypto key generate rsa modulus 2048
ip ssh version 2
access-list 99 permit 10.10.99.0 0.0.0.255
line vty 0 4
 access-class 99 in
 transport input ssh
 login local
line console 0
 logging synchronous
end
write memory
