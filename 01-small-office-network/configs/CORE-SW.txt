! CORE-SW - Layer 3 core switch. Cisco 3560-24PS.
! Roles: VLANs, inter-VLAN routing, DHCP server, OSPF, inter-VLAN ACLs, SSH.
!
enable
configure terminal
hostname CORE-SW
no ip domain-lookup
ip domain-name office.lab
enable secret CHANGE_ME
username admin secret CHANGE_ME
service password-encryption
ip routing
!
! --- VLANs ---
vlan 10
 name STAFF
vlan 20
 name GUEST
vlan 30
 name SERVERS
vlan 99
 name MGMT
vlan 999
 name BLACKHOLE
!
! --- Routed uplink to R1 ---
interface GigabitEthernet0/1
 description Routed link to R1
 no switchport
 ip address 10.10.0.2 255.255.255.252
 no shutdown
!
! --- Trunk to SW-ACCESS ---
interface FastEthernet0/1
 description Trunk to SW-ACCESS
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,99
 no shutdown
!
! --- Server port ---
interface FastEthernet0/2
 description SERVER-PT
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
 no shutdown
!
! --- SVIs (default gateways) ---
interface Vlan10
 description STAFF gateway
 ip address 10.10.10.1 255.255.255.0
 ip access-group STAFF-IN in
 no shutdown
interface Vlan20
 description GUEST gateway
 ip address 10.10.20.1 255.255.255.0
 ip access-group GUEST-IN in
 no shutdown
interface Vlan30
 description SERVERS gateway
 ip address 10.10.30.1 255.255.255.0
 no shutdown
interface Vlan99
 description MGMT gateway
 ip address 10.10.99.1 255.255.255.0
 no shutdown
!
! --- DHCP ---
ip dhcp excluded-address 10.10.10.1 10.10.10.20
ip dhcp excluded-address 10.10.20.1 10.10.20.20
ip dhcp excluded-address 10.10.99.1 10.10.99.20
ip dhcp pool STAFF
 network 10.10.10.0 255.255.255.0
 default-router 10.10.10.1
 dns-server 10.10.30.10
ip dhcp pool GUEST
 network 10.10.20.0 255.255.255.0
 default-router 10.10.20.1
 dns-server 10.10.30.10
ip dhcp pool MGMT
 network 10.10.99.0 255.255.255.0
 default-router 10.10.99.1
 dns-server 10.10.30.10
!
! --- OSPF ---
router ospf 1
 router-id 2.2.2.2
 network 10.10.0.0 0.0.255.255 area 0
 passive-interface default
 no passive-interface GigabitEthernet0/1
!
! --- Inter-VLAN ACLs ---
! Guests: DHCP, DNS to the server, internet only. No access to internal subnets.
ip access-list extended GUEST-IN
 permit udp any eq bootpc any eq bootps
 permit udp 10.10.20.0 0.0.0.255 host 10.10.30.10 eq domain
 deny   ip 10.10.20.0 0.0.0.255 10.10.0.0 0.0.255.255
 permit ip 10.10.20.0 0.0.0.255 any
! Staff: everything except the management VLAN.
ip access-list extended STAFF-IN
 permit udp any eq bootpc any eq bootps
 deny   ip 10.10.10.0 0.0.0.255 10.10.99.0 0.0.0.255
 permit ip 10.10.10.0 0.0.0.255 any
!
! --- Management: SSH only, from the management VLAN ---
crypto key generate rsa modulus 2048
ip ssh version 2
access-list 99 permit 10.10.99.0 0.0.0.255
line vty 0 4
 access-class 99 in
 transport input ssh
 login local
line console 0
 logging synchronous
end
write memory
