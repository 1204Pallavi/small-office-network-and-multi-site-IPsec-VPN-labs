! SW-ACCESS - access switch. Cisco 2960-24TT.
! Roles: user ports, port security, PortFast + BPDU Guard, unused-port lockdown, SSH.
!
enable
configure terminal
hostname SW-ACCESS
no ip domain-lookup
ip domain-name office.lab
enable secret CHANGE_ME
username admin secret CHANGE_ME
service password-encryption
!
vlan 10
 name STAFF
vlan 20
 name GUEST
vlan 30
 name SERVERS
vlan 99
 name MGMT
vlan 999
 name BLACKHOLE
!
! --- Uplink trunk ---
interface GigabitEthernet0/1
 description Trunk to CORE-SW
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,99
 no shutdown
!
! --- Staff ports (port security) ---
interface range FastEthernet0/1 - 8
 description STAFF
 switchport mode access
 switchport access vlan 10
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation shutdown
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
!
! --- Guest AP ports ---
interface range FastEthernet0/9 - 12
 description GUEST-AP
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
!
! --- Management PC port ---
interface FastEthernet0/13
 description MGMT-PC
 switchport mode access
 switchport access vlan 99
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation shutdown
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
!
! --- Unused ports ---
interface range FastEthernet0/14 - 24
 description UNUSED
 switchport mode access
 switchport access vlan 999
 shutdown
!
! --- Management SVI ---
interface Vlan99
 description Management
 ip address 10.10.99.2 255.255.255.0
 no shutdown
ip default-gateway 10.10.99.1
!
! --- Management: SSH only, from the management VLAN ---
crypto key generate rsa modulus 2048
ip ssh version 2
access-list 99 permit 10.10.99.0 0.0.0.255
line vty 0 15
 access-class 99 in
 transport input ssh
 login local
line console 0
 logging synchronous
end
write memory
